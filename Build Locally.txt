# Create Image:
podman build -t kodi-atomic:latest .

# Create ISO:
mkdir -p ./output

podman run --rm -it --privileged \
--pull=newer \
-v $(pwd)/output:/output \
-v /run/podman/podman.sock:/var/run/docker.sock \
quay.io/centos-bootc/bootc-image-builder:latest \
--type iso \
--local \
localhost/kodi-atomic:latest


# Build from remote container:
mkdir -p ./output

sudo podman run --rm -it --privileged \
  --pull=newer \
  -v $(pwd)/output:/output \
  quay.io/centos-bootc/bootc-image-builder:latest \
  --type iso \
  ghcr.io/your-username/my-kodi-htpc:latest


## Output ISO Location:
ls -lh ./output/bootiso/install.iso
